A browser can let the user choose an image, optionally resize or compress it, then send its bytes to a server over HTTP.

```html
<input id="image" type="file" accept="image/*">
<script>
  document.querySelector("#image").addEventListener("change", async (event) => {
    const file = event.target.files[0];
    if (!file) return;

    const form = new FormData();
    form.append("image", file);

    const response = await fetch("/api/analyze-image", {
      method: "POST",
      body: form
    });

    if (!response.ok) throw new Error("Image upload failed");
    console.log(await response.json());
  });
</script>
```

The browser’s `File` object contains the image data and metadata. `FormData` sends it as a multipart upload; the server can then process it or pass it to a vision model. For Ollama, the server can convert the image to base64 if the chosen Ollama endpoint expects that. Keeping the Ollama connection and any credentials on the server is generally safer than calling it directly from the browser.

The `device-width` value in your HTML is for viewport sizing on mobile; it doesn’t affect image preparation or uploading.

A browser can select an image, read it as a data URL, remove the `data:image/...;base64,` prefix, and send the remaining base64 string in Ollama’s `images` array.

Add this input to your HTML:

Then handle the selection and request in JavaScript:

```js
document.querySelector("#send").addEventListener("click", async () => {
  const file = document.querySelector("#image").files[0];
  const prompt = document.querySelector("#prompt").value.trim();
  if (!file || !prompt) return;

  const dataUrl = await new Promise((resolve, reject) => {
    const reader = new FileReader();
    reader.onload = () => resolve(reader.result);
    reader.onerror = () => reject(reader.error);
    reader.readAsDataURL(file);
  });

  const base64Image = dataUrl.split(",")[1];

  const response = await fetch("http://localhost:11434/api/generate", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      model: "gemma3",
      prompt,
      images: [base64Image],
      stream: false
    })
  });

  if (!response.ok) {
    throw new Error(`Ollama error: ${await response.text()}`);
  }

  const data = await response.json();
  document.querySelector("#result").textContent = data.response;
});
```

document.querySelector("#send").addEventListener("click", async () => {
  const file = document.querySelector("#image").files[0];
  const prompt = document.querySelector("#prompt").value.trim();
  if (!file || !prompt) return;

  try {
    const dataUrl = await new Promise((resolve, reject) => {
      const reader = new FileReader();
      reader.onload = () => resolve(reader.result);
      reader.onerror = () => reject(reader.error);
      reader.readAsDataURL(file);
    });

    const response = await fetch("http://localhost:11434/api/generate", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        model: "gemma3",
        prompt,
        images: [dataUrl.split(",")[1]],
        stream: false
      })
    });

    if (!response.ok) throw new Error(await response.text());

    const data = await response.json();
    document.querySelector("#result").textContent = data.response;
  } catch (error) {
    document.querySelector("#result").textContent =
      error instanceof Error ? error.message : "Request failed";
